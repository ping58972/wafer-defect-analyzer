cmake_minimum_required(VERSION 3.20)
project(wafer_defect_analyzer LANGUAGES CXX)

add_library(wafer_map src/wafer_map.cpp)
target_include_directories(wafer_map PUBLIC include)
target_compile_features(wafer_map PUBLIC cxx_std_20)
set(CMAKE_CXX_EXTENSIONS OFF)
set_target_properties(wafer_map PROPERTIES CXX_EXTENSIONS OFF)

function(enable_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endfunction()

enable_warnings(wafer_map)
add_executable(wafer-inspect src/main.cpp)
target_link_libraries(wafer-inspect PRIVATE wafer_map)
enable_warnings(wafer-inspect)

include(CTest)
if(BUILD_TESTING)
    add_executable(wafer_map_tests tests/wafer_map_tests.cpp)
    target_link_libraries(wafer_map_tests PRIVATE wafer_map)
    enable_warnings(wafer_map_tests)
    foreach(test IN ITEMS counts connectivity boundaries empty malformed maximum)
        add_test(NAME ${test} COMMAND wafer_map_tests ${test})
    endforeach()
    add_test(NAME cli_sample COMMAND wafer-inspect ${CMAKE_CURRENT_SOURCE_DIR}/examples/sample.csv)
    set_tests_properties(cli_sample PROPERTIES PASS_REGULAR_EXPRESSION "clusters=3")
endif()
